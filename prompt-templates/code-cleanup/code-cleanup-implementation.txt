### Automation Scripts

#### Whitespace Cleanup Script
```javascript
// scripts/cleanup-whitespace.js
const fs = require('fs');
const path = require('path');

function cleanWhitespace(filePath) {
  let content = fs.readFileSync(filePath, 'utf8');
  
  // Remove trailing spaces
  content = content.replace(/[ \t]+$/gm, '');
  
  // Normalize line endings
  content = content.replace(/\r\n/g, '\n');
  
  // Ensure single newline at end
  content = content.replace(/\n+$/, '\n');
  
  fs.writeFileSync(filePath, content);
  console.log(`Cleaned: ${filePath}`);
}

function processDirectory(dirPath) {
  const files = fs.readdirSync(dirPath);
  
  files.forEach(file => {
    const filePath = path.join(dirPath, file);
    const stat = fs.statSync(filePath);
    
    if (stat.isDirectory()) {
      processDirectory(filePath);
    } else if (/\.(ts|tsx|js|jsx|json|md|css)$/.test(file)) {
      cleanWhitespace(filePath);
    }
  });
}

// Usage: node scripts/cleanup-whitespace.js [directory]
const targetDir = process.argv[2] || '.';
processDirectory(targetDir);
```

#### Import Organization Script
```javascript
// scripts/organize-imports.js
const fs = require('fs');
const path = require('path');

function organizeImports(filePath) {
  let content = fs.readFileSync(filePath, 'utf8');
  
  // Extract imports
  const importRegex = /import\s+.*?from\s+['"][^'"]+['"];?\n?/g;
  const imports = content.match(importRegex) || [];
  
  if (imports.length === 0) return;
  
  // Remove existing imports
  content = content.replace(importRegex, '');
  
  // Sort and deduplicate imports
  const sortedImports = [...new Set(imports)].sort();
  
  // Add organized imports at the top
  content = sortedImports.join('\n') + '\n\n' + content.trim();
  
  fs.writeFileSync(filePath, content);
  console.log(`Organized imports: ${filePath}`);
}
```
