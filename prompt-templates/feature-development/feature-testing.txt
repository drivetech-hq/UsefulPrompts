```typescript
    render(<FeatureComponent />)
    // Test implementation
  })
  
  it('displays error states appropriately', () => {
    render(<FeatureComponent hasError={true} />)
    // Test implementation
  })
})
```

### API Testing
```typescript
// Example API test structure using Playwright
import { test, expect } from '@playwright/test'

test.describe('Feature API', () => {
  test('creates new feature successfully', async ({ request }) => {
    const response = await request.post('/api/feature', {
      data: { name: 'Test Feature' }
    })
    expect(response.ok()).toBeTruthy()
  })
  
  test('validates input data correctly', async ({ request }) => {
    const response = await request.post('/api/feature', {
      data: { name: '' }
    })
    expect(response.status()).toBe(400)
  })
})
```

### E2E Testing
```typescript
// Example E2E test structure using Playwright
import { test, expect } from '@playwright/test'

test('complete feature workflow', async ({ page }) => {
  // Navigate to feature
  await page.goto('/feature')
  
  // Perform actions
  await page.fill('[data-testid="input"]', 'test data')
  await page.click('[data-testid="submit"]')
  
  // Verify results
  await expect(page.locator('[data-testid="success"]')).toBeVisible()
})
```
